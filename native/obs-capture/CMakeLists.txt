cmake_minimum_required(VERSION 3.24)
project(OpenCamObsCapture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(required OBS_SOURCE OBS_RUNTIME FFMPEG_HEADERS)
  if(NOT IS_DIRECTORY "${${required}}")
    message(FATAL_ERROR "${required} must be an existing pinned source/runtime directory")
  endif()
endforeach()
set(OBS_INSTALL_PREFIX "")
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE}/libobs/obsconfig.h.in" generated/obsconfig.h)
add_executable(OpenCam.ObsCapture main.cpp)
target_include_directories(OpenCam.ObsCapture PRIVATE "${OBS_SOURCE}/libobs" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${FFMPEG_HEADERS}" "${SIMDE_HEADERS}")
if(APPLE)
  enable_language(OBJCXX)
  set_source_files_properties(main.cpp PROPERTIES LANGUAGE OBJCXX)
  set(frameworks "${OBS_RUNTIME}/Contents/Frameworks")
  # Explicit metadata-only test target: no OBS startup or media/device capture.
  add_executable(OpenCam.ObsDisplayTests EXCLUDE_FROM_ALL ../../tests/native/obs-mac-display-tests.mm)
  target_include_directories(OpenCam.ObsDisplayTests PRIVATE "${OBS_SOURCE}/libobs" "${CMAKE_CURRENT_BINARY_DIR}/generated" "${FFMPEG_HEADERS}" "${SIMDE_HEADERS}")
  target_compile_options(OpenCam.ObsDisplayTests PRIVATE -UNDEBUG)
  foreach(target OpenCam.ObsCapture OpenCam.ObsDisplayTests)
    target_link_libraries(${target} PRIVATE "${frameworks}/libobs.framework/libobs" "${frameworks}/libavformat.dylib" "${frameworks}/libavcodec.dylib" "${frameworks}/libavutil.dylib" "-framework Cocoa" "-framework CoreGraphics")
    set_target_properties(${target} PROPERTIES BUILD_RPATH "${frameworks}" INSTALL_RPATH "@loader_path/Contents/Frameworks")
  endforeach()
else()
  target_compile_definitions(OpenCam.ObsCapture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(OpenCam.ObsCapture PRIVATE "${OBS_IMPORT_LIBS}/obs.lib" "${OBS_IMPORT_LIBS}/avformat.lib" "${OBS_IMPORT_LIBS}/avcodec.lib" "${OBS_IMPORT_LIBS}/avutil.lib" user32 dxgi)
  add_library(OpenCamObsDisplay MODULE windows-display-module.c
    "${OBS_SOURCE}/plugins/win-capture/duplicator-monitor-capture.c"
    "${OBS_SOURCE}/plugins/win-capture/cursor-capture.c")
  target_include_directories(OpenCamObsDisplay PRIVATE "${OBS_SOURCE}/libobs"
    "${CMAKE_CURRENT_BINARY_DIR}/generated" "${OBS_SOURCE}/plugins/win-capture"
    "${OBS_SOURCE}/libobs-winrt" "${OBS_SOURCE}/deps/w32-pthreads")
  target_compile_definitions(OpenCamObsDisplay PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_compile_features(OpenCamObsDisplay PRIVATE c_std_11)
  target_link_libraries(OpenCamObsDisplay PRIVATE "${OBS_IMPORT_LIBS}/obs.lib"
    "${OBS_IMPORT_LIBS}/w32-pthreads.lib" user32 gdi32)
  set_target_properties(OpenCamObsDisplay PROPERTIES OUTPUT_NAME "win-capture" PREFIX "")
  if(MSVC)
    target_compile_options(OpenCamObsDisplay PRIVATE /W4 /utf-8)
  endif()
endif()
if(MSVC)
  target_compile_options(OpenCam.ObsCapture PRIVATE /W4 /utf-8)
else()
  target_compile_options(OpenCam.ObsCapture PRIVATE -Wall -Wextra -Werror=return-type)
endif()
