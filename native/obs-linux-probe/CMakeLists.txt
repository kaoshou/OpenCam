cmake_minimum_required(VERSION 3.28)
project(OpenCamLinuxProbe LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
  message(FATAL_ERROR "Linux x86_64 diagnostic probe only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OBS REQUIRED IMPORTED_TARGET libobs)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb xcb-shm xcb-randr xcb-xfixes xcb-xinerama xcb-composite x11)
pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
pkg_check_modules(PW REQUIRED IMPORTED_TARGET libpipewire-0.3 gio-unix-2.0 libdrm wayland-client)
pkg_check_modules(X264 REQUIRED IMPORTED_TARGET x264)
find_package(OpenGL REQUIRED)
if(NOT EXISTS "${OBS_SOURCE}/libobs/obs.h" OR NOT IS_DIRECTORY "${OBS_RUNTIME}/lib")
  message(FATAL_ERROR "Expected validated pinned source and built private runtime")
endif()
add_library(probe-glad OBJECT "${OBS_SOURCE}/deps/glad/src/glad.c")
target_include_directories(probe-glad PUBLIC "${OBS_SOURCE}/deps/glad/include")
target_link_libraries(probe-glad PUBLIC OpenGL::GL ${CMAKE_DL_LIBS})
function(probe_module name directory)
  set(files ${ARGN})
  list(TRANSFORM files PREPEND "${OBS_SOURCE}/plugins/${directory}/")
  add_library(${name} MODULE ${files})
  target_link_libraries(${name} PRIVATE PkgConfig::OBS)
  set_target_properties(${name} PROPERTIES PREFIX "" BUILD_RPATH "${OBS_RUNTIME}/lib")
  install(TARGETS ${name} LIBRARY DESTINATION lib/obs-plugins)
  install(DIRECTORY "${OBS_SOURCE}/plugins/${directory}/data/" DESTINATION "share/obs/obs-plugins/${name}")
endfunction()
probe_module(linux-capture linux-capture linux-capture.c xcomposite-input.c xcursor-xcb.c xhelpers.c xshm-input.c)
target_link_libraries(linux-capture PRIVATE probe-glad PkgConfig::XCB)
probe_module(linux-pulseaudio linux-pulseaudio linux-pulseaudio.c pulse-input.c pulse-wrapper.c)
target_link_libraries(linux-pulseaudio PRIVATE PkgConfig::PULSE)
probe_module(linux-pipewire linux-pipewire linux-pipewire.c formats.c pipewire.c portal.c screencast-portal.c camera-portal.c)
target_link_libraries(linux-pipewire PRIVATE probe-glad PkgConfig::PW)
probe_module(obs-x264 obs-x264 obs-x264.c obs-x264-plugin-main.c)
target_sources(obs-x264 PRIVATE "${OBS_SOURCE}/shared/opts-parser/opts-parser.c")
target_include_directories(obs-x264 PRIVATE "${OBS_SOURCE}/shared/opts-parser")
target_link_libraries(obs-x264 PRIVATE PkgConfig::X264)
add_library(linux-aac MODULE linux-aac.c "${OBS_SOURCE}/plugins/obs-ffmpeg/obs-ffmpeg-audio-encoders.c")
target_link_libraries(linux-aac PRIVATE PkgConfig::OBS PkgConfig::AV)
set_target_properties(linux-aac PROPERTIES PREFIX "" BUILD_RPATH "${OBS_RUNTIME}/lib")
install(TARGETS linux-aac LIBRARY DESTINATION lib/obs-plugins)
install(DIRECTORY "${OBS_SOURCE}/plugins/obs-ffmpeg/data/" DESTINATION share/obs/obs-plugins/linux-aac)
add_executable(OpenCam.ObsLinuxProbe main.cpp)
target_include_directories(OpenCam.ObsLinuxProbe PRIVATE ../obs-capture)
target_link_libraries(OpenCam.ObsLinuxProbe PRIVATE PkgConfig::OBS PkgConfig::AV PkgConfig::XCB PkgConfig::PW)
set_target_properties(OpenCam.ObsLinuxProbe PROPERTIES BUILD_RPATH "${OBS_RUNTIME}/lib")
target_compile_options(OpenCam.ObsLinuxProbe PRIVATE -Wall -Wextra -Werror=return-type)
install(TARGETS OpenCam.ObsLinuxProbe RUNTIME DESTINATION bin)
